# MirrorOS — Python dependencies
# Install: bash scripts/setup-voice.sh   (does both steps below)
#   pip3 install -r requirements.txt
#   pip3 install --no-deps "openwakeword>=0.6.0"
# (Pi OS Bookworm+ also needs --break-system-packages on both — the script adds it.)

# Wake word detection (openWakeWord — open-source, no cloud, no AccessKey)
#
# openwakeword itself is deliberately NOT listed: on Linux it hard-requires
# tflite-runtime, which has no wheels for Python 3.12+ (the Pi runs 3.13), so
# listing it makes `pip3 install -r` fail for the whole file. It is installed
# with --no-deps instead, and its real runtime deps are listed here. Inference
# uses onnxruntime (WAKE_FRAMEWORK=onnx, the default) — tflite is unavailable.
onnxruntime>=1.10.0
numpy>=1.24
tqdm>=4.0
scipy>=1.3
scikit-learn>=1.0

# Audio I/O
pyaudio>=0.2.13

# Speech-to-text (Whisper — transcribe.py loads the 'base' model)
openai-whisper>=20231117
# Used by transcribe.py to boost quiet audio before STT
soundfile>=0.12

# HTTP requests (used by wakeword.py to call backend)
requests>=2.31.0

# Text-to-speech — Piper (neural, offline, natural) is primary.
# Download a voice once:  python3 -m piper.download_voices en_US-amy-medium --download-dir server/voice/piper-voices
piper-tts>=1.4.0
# pyttsx3 (espeak) kept only as a fallback if Piper/voice is missing
pyttsx3>=2.90
